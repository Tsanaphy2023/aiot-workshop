# MicroPython Script for LEQs-AIoT v.1 (Deep+ AIoT 2027)
# Real-time Synchronized Driver with ESP32 I2C & Agronomy Engine

from machine import Pin, I2C, ADC
import time, math, ujson

# I2C Bus Configuration (SDA=GPIO21, SCL=GPIO22)
i2c = I2C(0, scl=Pin(22), sda=Pin(21), freq=400000)

# Actuator: Solenoid Valve / Relay on Configurable 1 (GPIO 16)
relay_valve = Pin(16, Pin.OUT)
relay_valve.value(0)

# Soil Tensiometer Sensor on Configurable 2 (GPIO 19 / ADC)
soil_adc = ADC(Pin(19))
soil_adc.atten(ADC.ATTN_11DB) # 0 - 3.6V Range

def calculate_vpd(temp_c, rh_percent):
    # Tetens equation for saturation vapor pressure (kPa)
    vp_sat = 0.61078 * math.exp((17.27 * temp_c) / (temp_c + 237.3))
    vp_actual = vp_sat * (rh_percent / 100.0)
    return vp_sat - vp_actual

while True:
    # Read simulated telemetry stream
    temp = 28.5  # SHT3x Sensor
    humidity = 65.0  # SHT3x Sensor
    pressure = 1013.2  # BMP280 Sensor
    lux = 650  # BH1750 Sensor
    soil_tension = 32.0  # Tensiometer (kPa)

    vpd = calculate_vpd(temp, humidity)

    # Automatic Irrigation Control Logic
    if soil_tension > 45.0:
        relay_valve.value(1) # Open Water Valve
        valve_status = "OPEN"
    else:
        relay_valve.value(0) # Close Valve
        valve_status = "CLOSED"

    telemetry = {
        "device": "LEQs-AIoT-v1",
        "temperature": temp,
        "humidity": humidity,
        "pressure": pressure,
        "lux": lux,
        "soil_tension_kpa": soil_tension,
        "vpd_kpa": round(vpd, 2),
        "solenoid_valve": valve_status
    }
    print("[TELEMETRY]", ujson.dumps(telemetry))
    time.sleep(1)
